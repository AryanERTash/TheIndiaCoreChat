import os
import asyncio
import json
from urllib.parse import unquote
from fastapi import FastAPI, Query, Request
from fastapi.responses import JSONResponse, FileResponse
from fastapi.middleware.cors import CORSMiddleware
from fastapi.middleware.gzip import GZipMiddleware
from starlette.middleware.base import BaseHTTPMiddleware
from pydantic import BaseModel
from dotenv import load_dotenv

# SlowAPI
from slowapi import Limiter
from slowapi.middleware import SlowAPIMiddleware
from slowapi.util import get_remote_address
from slowapi.errors import RateLimitExceeded

# Gemini LangChain
from langchain_google_genai import ChatGoogleGenerativeAI
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.messages import HumanMessage, AIMessage

# Load ENV
load_dotenv()
API_KEY = os.getenv("GEMINI_API_KEY")
if not API_KEY:
    raise RuntimeError("Missing GEMINI_API_KEY in .env")

MODEL = "gemini-2.5-flash-lite"
SYSTEM_PROMPT = "You are a helpful assistant. Reply in Markdown format."

# Base working directory (absolute path)
BASE_DIR = os.path.dirname(os.path.abspath(__file__))

# ─────────────────────────────────────────────
# FastAPI App
# ─────────────────────────────────────────────
app = FastAPI(title="Unified Gemini Server", version="1.0")


# ─────────────────────────────────────────────
# Serve index.html at /server
# ─────────────────────────────────────────────
@app.get("/server")
async def serve_root():
    index_path = os.path.join(BASE_DIR, "index.html")
    return FileResponse(index_path)


# ─────────────────────────────────────────────
# Serve any file or folder relative to project root
# Example:
#   /server/assets/js/chat.js
#   /server/test.html
#   /server/verify-implementation.js
# ─────────────────────────────────────────────
@app.get("/server/{path:path}")
async def serve_any(path: str):
    full_path = os.path.join(BASE_DIR, path)

    if os.path.exists(full_path) and os.path.isfile(full_path):
        return FileResponse(full_path)

    return JSONResponse({"error": "file not found"}, status_code=404)


# ─────────────────────────────────────────────
# Middleware setup
# ─────────────────────────────────────────────
limiter = Limiter(key_func=get_remote_address, default_limits=["10/minute"])
app.state.limiter = limiter
app.add_middleware(SlowAPIMiddleware)

app.add_exception_handler(
    RateLimitExceeded,
    lambda r, e: JSONResponse({"success": False, "error": "rate limit exceeded"})
)

app.add_middleware(GZipMiddleware, minimum_size=500)

app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],
    allow_credentials=True,
    allow_methods=["GET"],
    allow_headers=["*"],
)

class SecureHeadersMiddleware(BaseHTTPMiddleware):
    async def dispatch(self, request, call_next):
        res = await call_next(request)
        res.headers["X-Frame-Options"] = "DENY"
        res.headers["X-Content-Type-Options"] = "nosniff"
        res.headers["X-XSS-Protection"] = "1; mode=block"
        return res

app.add_middleware(SecureHeadersMiddleware)


# ─────────────────────────────────────────────
# Response schema
# ─────────────────────────────────────────────
class ChatResponse(BaseModel):
    success: bool
    error: str | None = None
    response: str | None = None


def format_error(msg: str):
    return JSONResponse(ChatResponse(success=False, error=msg).model_dump())


async def run_with_timeout(awaitable, timeout_sec=20):
    try:
        return await asyncio.wait_for(awaitable, timeout=timeout_sec)
    except asyncio.TimeoutError:
        return "TIMEOUT"


# ─────────────────────────────────────────────
# Gemini Chat Endpoint
# ─────────────────────────────────────────────
@app.get("/server/api/chat", response_model=ChatResponse)
@limiter.limit("50/minute")
async def chat(
    request: Request,
    q: str = Query(...),
    sys: str = Query(None),
    history: str = Query(None)
):
    try:
        system_prompt = sys or SYSTEM_PROMPT

        conversation = []
        if history:
            try:
                arr = json.loads(unquote(history))
                for m in arr:
                    if m["role"] == "user":
                        conversation.append(HumanMessage(content=m["content"]))
                    elif m["role"] == "assistant":
                        conversation.append(AIMessage(content=m["content"]))
            except:
                pass

        messages = [("system", system_prompt)]
        for m in conversation:
            if isinstance(m, HumanMessage):
                messages.append(("human", m.content))
            else:
                messages.append(("ai", m.content))

        messages.append(("human", "{user_input}"))

        prompt = ChatPromptTemplate.from_messages(messages)

        llm = ChatGoogleGenerativeAI(api_key=API_KEY, model=MODEL, temperature=0.2)
        chain = prompt | llm

        result = await run_with_timeout(chain.ainvoke({"user_input": q}), 25)

        if result == "TIMEOUT":
            return format_error("timeout")

        return ChatResponse(success=True, response=result.content)

    except Exception as e:
        print("chat error:", e)
        return format_error("internal error")


# ─────────────────────────────────────────────
# Health endpoint
# ─────────────────────────────────────────────
@app.get("/server/api/health")
async def health():
    return {"status": "ok"}
